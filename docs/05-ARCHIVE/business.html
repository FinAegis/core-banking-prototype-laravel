Code Example

// Initialize payment const payment = await gcu.payments.create({ amount: 1000.00, currency: 'USD', recipient: 'merchant@example.com', reference: 'ORDER-12345' }); // Check payment status const status = await gcu.payments.get(payment.id); console.log(status.state); // 'completed' // Get real-time GCU rate const rate = await gcu.rates.current(); console.log(rate.gcu_usd); // 1.0234
